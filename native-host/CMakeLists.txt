cmake_minimum_required(VERSION 3.24)
project(MojaveIsoNeuralHost LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "The neural helper requires Windows x64")
endif()
set(NGX_SDK "" CACHE PATH "NVIDIA/DLSS SDK checkout (not redistributed)")
if(NOT EXISTS "${NGX_SDK}/include/nvsdk_ngx.h")
 message(FATAL_ERROR "Set NGX_SDK to the NVIDIA DLSS SDK checkout")
endif()
add_executable(MojaveIsoNeuralHost ../native-plugin/third-party/dlss5-feeder/host/dlss5-feed-host64.cpp)
target_compile_features(MojaveIsoNeuralHost PRIVATE cxx_std_17)
target_include_directories(MojaveIsoNeuralHost PRIVATE "${NGX_SDK}/include")
target_compile_options(MojaveIsoNeuralHost PRIVATE /O2 /EHsc)
target_link_libraries(MojaveIsoNeuralHost PRIVATE "${NGX_SDK}/lib/Windows_x86_64/x64/nvsdk_ngx_d.lib" version winmm kernel32 user32 gdi32 advapi32 ole32)
